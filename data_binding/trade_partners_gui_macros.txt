macro = {
	description = "Get imports (units) from root to target market"
	definition = "TradePartnersGetImportsFrom(Importer, Exporter)"
	replace_with = "GuiScope.SetRoot(Importer.MakeScope).AddScope('base_market', Exporter.MakeScope).ScriptValue('trade_partners_total_imports_from_target')"
}

macro = {
	description = "Get exports (units) from root to target market"
	definition = "TradePartnersGetExportsTo(Exporter, Importer)"
	replace_with = "GuiScope.SetRoot(Exporter.MakeScope).AddScope('base_market', Importer.MakeScope).ScriptValue('trade_partners_total_exports_to_target')"
}

macro = {
	description = "Get monetary export value (£ at World Market Price) from Exporter to Importer"
	definition = "TradePartnersGetExportValueTo(Exporter, Importer)"
	replace_with = "GuiScope.SetRoot(Exporter.MakeScope).AddScope('base_market', Importer.MakeScope).ScriptValue('trade_partners_export_value_to_target')"
}

macro = {
	description = "Get monetary import value (£ at World Market Price) of Importer from Exporter (symmetric with Exporter -> Importer)"
	definition = "TradePartnersGetImportValueFrom(Importer, Exporter)"
	replace_with = "GuiScope.SetRoot(Exporter.MakeScope).AddScope('base_market', Importer.MakeScope).ScriptValue('trade_partners_export_value_to_target')"
}

macro = {
	description = "True if root market imports from target market"
	definition = "TradePartnersIsImporterFrom(Importer, Exporter)"
	replace_with = "GreaterThanOrEqualTo_CFixedPoint(TradePartnersGetImportsFrom(Importer, Exporter), '(CFixedPoint)1')"
}

macro = {
	description = "True if root market exports to target market"
	definition = "TradePartnersIsExporterTo(Exporter, Importer)"
	replace_with = "GreaterThanOrEqualTo_CFixedPoint(TradePartnersGetExportsTo(Exporter, Importer), '(CFixedPoint)1')"
}

macro = {
	description = "True if two markets have bilateral trade >= 1 in either direction"
	definition = "TradePartnersHasAnyTrade(MarketA, MarketB)"
	replace_with = "Or(TradePartnersIsImporterFrom(MarketA, MarketB), TradePartnersIsExporterTo(MarketA, MarketB))"
}

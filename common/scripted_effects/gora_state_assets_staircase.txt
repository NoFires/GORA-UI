# Per-building national value (cost tier x 250 x levels)
# Building scope. Computes ROOT's nationally-owned value for this building.
# gora_sa_nat_levels_tmp must be set first by the binary-tree effect above.
# The result stays in the effect-local gora_sa_nat_value_tmp variable.
gora_sa_compute_nat_value_for_building_effect = {
	if = {
		limit = { is_building_group = bg_trade }
		set_local_variable = {
			name = gora_sa_nat_value_tmp
			value = { value = 100 multiply = 250 multiply = local_var:gora_sa_nat_levels_tmp }
		}
	}
	else_if = {
		limit = {
			OR = {
				is_building_group = bg_agriculture
				is_building_group = bg_ranching
				is_building_group = bg_plantations
				is_building_group = bg_logging
				is_building_group = bg_rubber
				is_building_group = bg_fishing
				is_building_group = bg_whaling
			}
		}
		set_local_variable = {
			name = gora_sa_nat_value_tmp
			value = { value = 200 multiply = 250 multiply = local_var:gora_sa_nat_levels_tmp }
		}
	}
	else_if = {
		# Railway must precede bg_private_infrastructure medium tier.
		limit = { is_building_type = building_railway }
		set_local_variable = {
			name = gora_sa_nat_value_tmp
			value = { value = 800 multiply = 250 multiply = local_var:gora_sa_nat_levels_tmp }
		}
	}
	else_if = {
		limit = {
			OR = {
				is_building_group = bg_mining
				is_building_group = bg_oil_extraction
				is_building_group = bg_private_infrastructure
				is_building_group = bg_arts
				is_building_group = bg_power
			}
		}
		set_local_variable = {
			name = gora_sa_nat_value_tmp
			value = { value = 400 multiply = 250 multiply = local_var:gora_sa_nat_levels_tmp }
		}
	}
	else_if = {
		limit = { is_building_group = bg_light_industry }
		set_local_variable = {
			name = gora_sa_nat_value_tmp
			value = { value = 600 multiply = 250 multiply = local_var:gora_sa_nat_levels_tmp }
		}
	}
	else = {
		# Heavy industry, military industry, and anything else.
		set_local_variable = {
			name = gora_sa_nat_value_tmp
			value = { value = 800 multiply = 250 multiply = local_var:gora_sa_nat_levels_tmp }
		}
	}

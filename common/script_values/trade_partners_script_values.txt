# Reconstructs the current World Market Price (`Goods.GetWorldMarketBasePrice`) for an
# exporter's `market_goods` scope:
#   demand = goods.world_market_imports + traded_quantity * 10 (WORLD_MARKET_GOODS_DEMAND_BUFFER)
#   supply = goods.world_market_exports
#   mult   = clamp(1 + 0.75 * clamp((demand - supply) / max(min(demand, supply), 1), -1, 1), 0.25, 1.75)
#   if exporter holds > 50% of global exports: mult *= (1 + market_goods_export_share * 0.2) (MONOPOLY_PRICE_IMPACT)
# Note: in Victoria 3 script_values, `min = X` floors at X (max(v, X)) and `max = X` caps at X (min(v, X)).
trade_partners_world_market_goods_price = {
	value = goods.world_market_imports
	add = trade_partners_goods_demand_buffer
	save_temporary_value_as = tp_wm_demand

	value = goods.base_price
	multiply = {
		value = 1
		add = {
			value = scope:tp_wm_demand
			subtract = goods.world_market_exports
			divide = {
				value = scope:tp_wm_demand
				max = goods.world_market_exports
				min = 1
			}
			min = -1
			max = 1
			multiply = 0.75
		}
		if = {
			limit = { market_goods_export_share > 0.5 }
			multiply = {
				value = 1
				add = {
					value = market_goods_export_share
					multiply = 0.2
				}
			}
		}
		min = 0.25
		max = 1.75
	}
}

# Flat, no-rollback binary search (0..65535.5, 0.5-unit precision) recovering the exact
# per-good export volume from root `market_goods`'s market to `scope:tp_target_market`.
# Re-binds `scope:tp_good` locally and keeps all `save_temporary_value_as` at top level
# with unique names per bit (no nested `if` saves, no temp-name reuse).
trade_partners_goods_export_amount = {
	value = 0
	if = { limit = { goods = { save_temporary_scope_as = tp_good } } add = 0 }
	if = { limit = { market = { market_exports = { target = scope:tp_target_market goods = scope:tp_good value >= 32768 } } } add = 32768 }
	save_temporary_value_as = tp_e15

	save_temporary_value_as = tp_e14

# Monetary value (£) of a single exported `market_goods` to `scope:tp_target_market`.
trade_partners_goods_export_value = {
	value = trade_partners_goods_export_amount
	multiply = trade_partners_world_market_goods_price
}

# Core monetary sum (£) of all goods (>= 1 unit) exported from `this` market to `scope:tp_target_market`.
trade_partners_export_value_core = {
	value = 0
	every_market_goods = {
		limit = {
			market_goods_exports >= 1
			goods = { save_temporary_scope_as = tp_good }
			market = { market_exports = { target = scope:tp_target_market goods = scope:tp_good value >= 1 } }
		}
		add = trade_partners_goods_export_value
	}
}

# Total monetary value (£) of all goods (>= 1 unit) exported from root market
# to scope:base_market, priced at each good's current World Market Price.
# Also used symmetrically by TradePartnersGetImportValueFrom(Importer, Exporter).
trade_partners_export_value_to_target = {
	value = 0
	if = {
		limit = { scope:base_market = { save_temporary_scope_as = tp_target_market } }
		add = trade_partners_export_value_core
	}
}

# Symmetric alias for callers that pass root = Importer, scope:base_market = Exporter
# (avoids shadowing permanent `base_market` and keeps live hot-reload compatible).
trade_partners_import_value_from_target = {
	value = 0
	if = {
		limit = { save_temporary_scope_as = tp_target_market }
		scope:base_market = {
			add = trade_partners_export_value_core
		}
	}
}

# =============================================================================
# Improved ParliamentGraph - Scripted GUIs (v3.1)
# -----------------------------------------------------------------------------
# 1. `ipg_capture_election_vote_share_sg`: Party -> Country bridge used by
#    `gui/ipg_election_capture.gui` to persist GUI-only election vote shares.
# 2. `ipg_open_refresh_sg`: Country-scoped on-demand refresh invoked when
#    `parliament_item` opens/expands; refreshes dynamic seats + auto layout and
#    triggers a one-shot election capture on mid-save loads if needed.
# 3. `ipg_set_formula_*_sg`: In-game live apportionment formula switches
#    (Auto / Hare / D'Hondt / Sainte-Lague / 5% Threshold / Mild FPTP).
# 4. `ipg_inspect_foreign_sg`: On-demand Power + Alignment calculation for any
#    foreign AI country viewed on the diplomatic Country Politics panel.
# =============================================================================

ipg_capture_election_vote_share_sg = {
	scope = party
	saved_scopes = {
		result
	}
	effect = {
		random_member = {
			save_scope_as = ipg_capture_member
		}
		save_scope_as = ipg_capture_party
		scope:ipg_capture_member.owner = {
			ipg_election_vote_setter = yes
		}
	}
}

ipg_open_refresh_sg = {
	scope = country
	effect = {
		ipg_recalculate_dynamic_structures = yes
		if = {
			limit = {
				NOT = { has_variable_list = ipg_electoral_parties }
				NOT = { has_variable = ipg_election_capture_requested }
				any_active_party = {
					party_has_members = yes
				}
			}
			ipg_request_election_capture = yes
		}
	}
}

ipg_set_formula_auto_sg = {
	scope = country
	effect = {
		remove_variable = ipg_formula_override
		ipg_recalculate_dynamic_structures = yes
	}
}

ipg_set_formula_1_sg = {
	scope = country
	effect = {
		set_variable = { name = ipg_formula_override value = 1 }
		ipg_recalculate_dynamic_structures = yes
	}
}

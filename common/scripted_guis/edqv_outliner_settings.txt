# Display preference is saved on the country; no economy values are changed.
edqv_outliner_is_disabled = {
	scope = country
	is_shown = { has_variable = edqv_outliner_disabled }
}

edqv_toggle_outliner = {
	scope = country
	is_shown = { is_player = yes }
	is_valid = { is_player = yes }
	effect = {
		if = {
			limit = { has_variable = edqv_outliner_disabled }
			remove_variable = edqv_outliner_disabled
		}
		else = {
			set_variable = { name = edqv_outliner_disabled value = 1 }
		}
	}
}

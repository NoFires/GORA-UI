# Observer-mode refresh hooks for Trade Partners.
# In normal player sessions, partner charts refresh on demand via Scripted GUI and these
# on_actions exit immediately on the first trigger check (`NOT = { any_country = { is_player = yes } }`).

on_monthly_pulse = {
	on_actions = {
		on_actions_trade_partners_market_update
	}
}

on_game_started_after_lobby = {
	on_actions = {
		on_actions_trade_partners_market_update
	}
}

on_actions_trade_partners_market_update = {
	effect = {
		trade_partners_market_update_observer = yes
	}
}
